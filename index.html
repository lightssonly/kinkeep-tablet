Type /984619 show"); } listen(integer channel, string speaker, key avatarId, string message) { string cmd = llToLower(llStringTrim(message, STRING_TRIM)); integer page = 1; if (cmd != "show") { if (llGetSubString(cmd, 0, 4) != "show ") return; string textPage = llStringTrim(llGetSubString(cmd, 5, -1), STRING_TRIM); page = (integer)textPage; if (page < 1 || page > 50 || (string)page != textPage) { llOwnerSay("KinKeep: Please use /984619 show or /984619 show 2"); return; } } if (pending != NULL_KEY) { llOwnerSay("KinKeep: Screen update already in progress."); return; } mediaFace = findScreenFace(); if (mediaFace < 0) { llOwnerSay("KinKeep: I couldn't find the existing KinKeep webpage on this prim's media face. No changes made. Ask for help identifying the screen prim."); return; } string credential = llLinksetDataRead(SECRET_KEY); string tabletId = llLinksetDataRead(TABLET_KEY); if (llStringLength(credential) != 64 || llStringLength(tabletId) != 36) { llOwnerSay("KinKeep: Pairing data missing. Stop here."); return; } string payload = llList2Json(JSON_OBJECT, [ "tablet_id", tabletId, "device_credential", credential, "page", page ]); pending = llHTTPRequest(ROSTER_URL, [HTTP_METHOD, "POST", HTTP_MIMETYPE, "application/json"], payload); if (pending == NULL_KEY) { llOwnerSay("KinKeep: Couldn't begin screen update."); return; } currentPage = page; llSetTimerEvent(45.0); llOwnerSay("KinKeep: Fetching family profiles for the screen..."); } http_response(key reqId, integer code, list headers, string body) { if (reqId != pending) return; pending = NULL_KEY; llSetTimerEvent(0.0); if (code != 200 || llJsonGetValue(body, ["ok"]) != JSON_TRUE) { string err = llJsonGetValue(body, ["error"]); if (err == "" || err == JSON_NULL || err == JSON_INVALID) err = "Could not load roster"; llOwnerSay("KinKeep: Screen update failed (HTTP " + (string)code + "): " + err); return; } // The REST response contains only public-on-screen names, roles and emojis. // NEVER append linkset device secrets, UUIDs or PINs to the screen URL. string screenUrl = SITE + "#kkroster=" + llStringToBase64(body); if (llStringLength(screenUrl) > 1024) { llOwnerSay("KinKeep: This roster page is too long for the MOAP URL. Nothing changed."); return; } integer result = llSetLinkMedia(LINK_THIS, mediaFace, [PRIM_MEDIA_CURRENT_URL, screenUrl]); if (result == STATUS_OK) llOwnerSay("KinKeep: Live roster sent to tablet screen, page " + (string)currentPage + ". Open the Chores app."); else llOwnerSay("KinKeep: Couldn't update media URL. Status " + (string)result + ". Original display unchanged."); } timer() { pending = NULL_KEY; llSetTimerEvent(0.0); llOwnerSay("KinKeep: Screen update timed out. Existing display unchanged."); } }
